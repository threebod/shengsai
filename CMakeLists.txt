cmake_minimum_required(VERSION 3.22)
project(shengsai VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_subdirectory(upper_computer)
if(BUILD_TESTING AND NOT ANDROID)
    find_package(Qt6 REQUIRED COMPONENTS Test SerialPort Quick QuickControls2)
    add_executable(test_controller tests/test_controller.cpp upper_computer/Controller.cpp upper_computer/Controller.h)
    set_target_properties(test_controller PROPERTIES AUTOMOC ON)
    target_link_libraries(test_controller PRIVATE Qt6::Test Qt6::SerialPort)
    add_test(NAME controller COMMAND test_controller)
    add_executable(test_layout tests/test_layout.cpp upper_computer/Controller.cpp upper_computer/Controller.h)
    set_target_properties(test_layout PROPERTIES AUTOMOC ON WIN32_EXECUTABLE OFF)
    target_compile_definitions(test_layout PRIVATE SHENGSAI_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(test_layout PRIVATE Qt6::Test Qt6::Quick Qt6::QuickControls2 Qt6::SerialPort)
    add_test(NAME layout COMMAND test_layout)
    set_tests_properties(layout PROPERTIES ENVIRONMENT "QT_QUICK_BACKEND=software;QT_QPA_PLATFORM=offscreen")
    if(WIN32)
        set_tests_properties(layout PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<SHELL_PATH:$<TARGET_FILE_DIR:Qt6::Core>>")
    endif()
    add_executable(test_can_mailbox tests/test_can_mailbox.c firmware/jog_can.c)
    target_include_directories(test_can_mailbox PRIVATE firmware/tests/stubs)
    add_test(NAME can_mailbox COMMAND test_can_mailbox)
    foreach(name protocol axis_feedback mechanism_runtime scale_store obstacle_turret)
        add_executable(test_${name} tests/test_${name}.c)
        add_test(NAME ${name} COMMAND test_${name})
    endforeach()
    foreach(name route ring_poses qr_scan vision ring_store gripper_store tjc_obstacles)
        add_executable(test_${name} firmware/tests/test_${name}.c)
        target_link_libraries(test_${name} PRIVATE m)
        add_test(NAME ${name} COMMAND test_${name})
    endforeach()
endif()
